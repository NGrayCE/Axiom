cmake_minimum_required(VERSION 3.10)
project(Axiom C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED True)

# --- 1. Core Engine Library ---
set(AXIOM_HEADERS
    include/ax_math.h
    include/ax_core.h
    include/ax_platform.h
    include/ax_graphics.h
	include/ax_ui.h
	include/ax_assets.h
)

set(AXIOM_SOURCES
    src/core/ax_arena.c
    src/core/ax_engine.c
    src/graphics/ax_graphics.c
	src/ui/ax_ui.c
	src/assets/ax_assets.c
)

add_library(axiom_core STATIC ${AXIOM_HEADERS} ${AXIOM_SOURCES})
target_include_directories(
						axiom_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include
						${CMAKE_CURRENT_SOURCE_DIR}/vendor/stb)

# --- 2. Headless Test Bench ---
add_executable(axiom_headless src/platform/headless_main.c)
target_link_libraries(axiom_headless PRIVATE axiom_core)

# --- 3. SDL3 Host Application ---
set(SDL3_DIR ${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL3)
set(SDL3_INCLUDE_DIR ${SDL3_DIR}/include)
set(SDL3_LIB_DIR ${SDL3_DIR}/lib/x64)

add_executable(axiom_host_sdl src/platform/sdl_main.c)

target_include_directories(axiom_host_sdl PRIVATE ${SDL3_INCLUDE_DIR})

# Link the Axiom Core and the SDL3 library
target_link_libraries(axiom_host_sdl PRIVATE axiom_core ${SDL3_LIB_DIR}/SDL3.lib)

# Automatically copy SDL3.dll to the build folder
add_custom_command(TARGET axiom_host_sdl POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${SDL3_LIB_DIR}/SDL3.dll"
    $<TARGET_FILE_DIR:axiom_host_sdl>
)
